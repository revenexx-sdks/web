    id: '', // optional
    locationId: '', // optional
    productId: '', // optional

    onHand: 42, // optional
    reserved: 5, // optional
    reorderPoint: 10, // optional
    metadata: '{}', // optional
    createdAt: '2026-01-01T12:00:00Z', // optional
